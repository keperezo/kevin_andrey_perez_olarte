Calcula la media, mínimo, máximo y varianza (en SQL) de la duración en milisegundos de todas las pistas.

resultado <- dbGetQuery(con, "
  SELECT
    ROUND(AVG(t.Milliseconds),  2)   AS duracion_media_min,
    ROUND(MIN(t.Milliseconds),  2)   AS duracion_min_min,
    ROUND(MAX(t.Milliseconds),  2)   AS duracion_max_min,
    ROUND(AVG(t.UnitPrice),             2)   AS precio_medio,
    ROUND( AVG(t.Milliseconds * t.Milliseconds) - AVG(t.Milliseconds) * AVG(t.Milliseconds),0)  AS varianza_ms
  FROM     Track t;
")

resultado_9 <- dbGetQuery(con, "
